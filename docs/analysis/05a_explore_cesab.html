Author

Romain Frelat

Published

The objective is to explore the diversity of FRB CESAB projects through network analysis. The exploration contains three parts:

  1. membership network
  2. citation coupling network

Project network

Based on the number of shared members, we can build a network of projects.

Code 
#netPR2 <- delete_vertices(netPR, degree(netPR) == 1)
modPR <- cluster_fast_greedy(netPR)
cluPR <- membership(modPR)
nclu <- table(cluPR)
cclu <- colorspace::qualitative_hcl(n = length(nclu), palette = "Dark 3")
cclu[nclu == 1] <- "#808080" # grey
colPR <- cclu[cluPR]
simclu <- ifelse(cluPR <= (sum(nclu > 1)), cluPR, "isolated")

The network has 74 nodes (one node = one project) and 125 edges (one edge = shared reference between two projects). The connectance is 0.046 and the modularity 0.53

2. Citation coupling

# missing the association reference <-> project ...
# ref <- read.csv("data/raw-data/cesab/Zotero_FRB-CESAB.csv")

M <- readRDS(file.path(out_data, "cesab_bibliometrix.rds"))
M$shortjournal <- abbreviate(M$SO, 10)
M$shorttitle <- substr(M$display_name, 1, 30)
# get the co-citation network of references
NetMatrix <- bibliometrix::biblioNetwork(
  M,
  analysis = "coupling",
  network = "references",
  n = NULL,
  short = FALSE,
  sep = ";"
)
dim(NetMatrix) # 395
[1] 395 395
Code

# 46722 references with 627 original references

net1 <- igraph::graph_from_adjacency_matrix(
  NetMatrix,
  mode = "undirected",
  diag = FALSE,
  weighted = TRUE
)
V(net1)$name <- M$shortname[match(V(net1)$name, M$SR)]

mod1 <- cluster_fast_greedy(net1)


# delete non connected literature
# net2 <- delete_vertices(net1, degree(net1) == 0)
net2 <- delete_vertices(net1, components(net1)$membership != 1)

# modularity
mod2 <- cluster_fast_greedy(net2)
clu2 <- membership(mod2)
sclu2 <- ifelse(clu2 <= (sum(table(clu2) > 1)), clu2, "isolated")
sclu2 <- as.factor(sclu2)

cclu[levels(sclu2) == "isolated"] <- "#808080" # grey
