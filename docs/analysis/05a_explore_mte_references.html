# remove duplicated : from 1047 to 1035 rows
M <- M[!duplicated(M$DI), ]
# length(M$DI) # 1035

Last1 <- sapply(strsplit(M$AU_CORR, " "), function(x) x[[length(x)]])
M$shortname <- paste(Last1, M$PY, sep = "_")
dup <- duplicated(M$shortname)
n <- 2
while (sum(dup) > 0) {
  M$shortname[dup] <- paste(M$shortname[dup], n, sep = "_")
  dup <- duplicated(M$shortname)
  n <- n + 1
}

# M$AU[M$AU_FLAG]
NetMatrix <- bibliometrix::biblioNetwork(
  M,
  analysis = "co-citation",
  network = "references",
  n = NULL,
  short = TRUE,
  sep = ";"
)
# how to replace names by readable names
# dimnames(NetMatrix)

# select only primary references
keep <- dimnames(NetMatrix)[[1]] %in% M$id_oa
NetMatrix <- NetMatrix[keep, keep]
#fmt: skip
dimnames(NetMatrix)[[1]] <- M$shortname[match(dimnames(NetMatrix)[[1]], M$id_oa)]
#fmt: skip
dimnames(NetMatrix)[[2]] <- M$shortname[match(dimnames(NetMatrix)[[2]], M$id_oa)]

netCOC <- igraph::graph_from_adjacency_matrix(
  NetMatrix,
  mode = "undirected",
  diag = FALSE,
  weighted = TRUE
